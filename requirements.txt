# --- Ядро CrewAI ---------------------------------------------------------
# crewai — основной фреймворк для оркестрации мульти-агентных систем
# (создание Agent, Task, Crew, декораторы @CrewBase и т.д.).
crewai[anthropic]>=0.86.0

# crewai-tools — набор готовых инструментов (Tools) для агентов CrewAI
# (поиск в интернете, работа с файлами и т.д.), а также базовый класс
# BaseTool, на основе которого мы пишем свой trade_tool.py.
crewai-tools>=0.17.0

# --- Telegram-клиент -------------------------------------------------------
# telethon — библиотека для подключения к Telegram по User API
# (позволяет читать сообщения из чатов/каналов от имени обычного аккаунта,
# в отличие от Bot API, где бот должен быть добавлен в чат админом).
telethon>=1.36.0

# --- Работа с биржами -------------------------------------------------------
# ccxt — унифицированная библиотека для подключения к 100+ криптобиржам

# create_order(), fetch_balance() работают для всех бирж.
ccxt>=4.3.0

# --- Переменные окружения ---------------------------------------------------
# python-dotenv — загружает переменные из файла .env в os.environ,
# чтобы секретные ключи не были зашиты в коде.
python-dotenv>=1.0.1

# --- Валидация данных --------------------------------------------------------
# pydantic используется CrewAI "под капотом" для валидации структур данных
# (например, JSON-результата парсинга сигнала). Указываем явно, чтобы версия
# была предсказуемой и совместимой с crewai.
pydantic>=2.7.0
